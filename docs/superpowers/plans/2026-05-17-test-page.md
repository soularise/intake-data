# IntakeData Test Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a single standalone `test.html` file that lets a user upload an eldercare document, hit the live IntakeData Vercel API, and see extracted fields with confidence scores — polished enough for demos.

**Architecture:** Single HTML file with Tailwind CSS and Alpine.js loaded from CDN. Three states managed in Alpine: upload form, results panel, error state. FileReader API encodes the file to base64 client-side before POSTing to `/v1/extract`.

**Tech Stack:** HTML5, Tailwind CSS v3 (CDN), Alpine.js v3 (CDN), Fetch API, FileReader API

  <script>
    function app() {
      return {
        apiUrl: 'https://intake-data-git-main-soularises-projects.vercel.app',
        docType: 'intake_form',
        file: null,
        fileName: '',
        fileSize: '',
        oversized: false,
        dragging: false,
        loading: false,
        result: null,
        error: null,
        showAllFields: false,

        KEY_FIELDS: {
          intake_form: [
            'patient_name', 'date_of_birth', 'gender', 'phone_primary',
            'insurance_provider', 'policy_number', 'subscriber_name', 'subscriber_relationship',
          ],
          insurance_card: [
            'insurance_company', 'member_name', 'member_id', 'group_number',
            'plan_type', 'rx_bin', 'customer_service_phone',
          ],
        },

        MIME_MAP: {
          pdf: 'application/pdf',
          png: 'image/png',
          jpg: 'image/jpeg',
          jpeg: 'image/jpeg',
          webp: 'image/webp',
        },

        handleFile(e) {
          const f = e.target.files[0];
          if (f) this.setFile(f);
        },

        handleDrop(e) {
          this.dragging = false;
          const f = e.dataTransfer.files[0];
          if (f) this.setFile(f);
        },

        setFile(f) {
          const ext = f.name.split('.').pop().toLowerCase();
          if (!this.MIME_MAP[ext]) {
            alert('Unsupported file type. Please upload a PDF, PNG, JPG, or WebP.');
            return;
          }
          this.file = f;
          this.fileName = f.name;
          this.fileSize = (f.size / 1024 / 1024).toFixed(2) + ' MB';
          this.oversized = f.size > 8 * 1024 * 1024;
        },

        async extract() {
          if (!this.file || this.loading) return;
          this.loading = true;
          this.result = null;
          this.error = null;

          try {
            const base64 = await new Promise((resolve, reject) => {
              const reader = new FileReader();
              reader.onload = e => resolve(e.target.result.split(',')[1]);
              reader.onerror = reject;
              reader.readAsDataURL(this.file);
            });

            const res = await fetch(
              this.apiUrl.replace(/\/$/, '') + '/v1/extract',
              {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ document_type: this.docType, file: base64 }),
              }
            );

            const data = await res.json();

            if (!res.ok || !data.success) {
              this.error = data.error || { code: 'HTTP_' + res.status, message: 'Request failed' };
            } else {
              this.result = data;
              this.showAllFields = false;
            }
          } catch (err) {
            this.error = { code: 'NETWORK_ERROR', message: err.message };
          } finally {
            this.loading = false;
          }
        },

        reset() {
          this.file = null;
          this.fileName = '';
          this.fileSize = '';
          this.oversized = false;
          this.result = null;
          this.error = null;
          this.showAllFields = false;
        },

        keyFields() {
          return this.KEY_FIELDS[this.result?.document_type || this.docType] || [];
        },

        allFieldsSorted() {
          if (!this.result?.fields) return [];
          return Object.entries(this.result.fields)
            .sort((a, b) => (b[1]?.confidence ?? 0) - (a[1]?.confidence ?? 0));
        },

        confidenceClass(score) {
          if (score == null) return 'bg-slate-300';
          if (score >= 0.8) return 'bg-green-500';
          if (score >= 0.5) return 'bg-amber-500';
          return 'bg-red-500';
        },
      };
    }
  </script>
