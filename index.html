# Install
pip install requests

# Extract a patient intake form
import requests

response = requests.post(
    "https://api.intakedata.com/v1/extract",
    headers={"Authorization": "Bearer YOUR_API_KEY"},
    json={
        "document_type": "intake_form",
        "file": "base64-encoded-pdf"
    }
)

data = response.json()
# data["fields"]["patient_name"]["value"]
